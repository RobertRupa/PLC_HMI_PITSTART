; I/O:
; X0  AUTOMATE_PRESENT / INVERTER_OK
; X1  PRACA
; X4  mechanical STOP
; X5  mechanical P1
; X6  mechanical P2
; X7  mechanical P3
; X10 mechanical P4
; X11 mechanical P5
; X12 mechanical P6
; X13 MANUAL / FREE
; X27 RM5 CH1
;
; Y0  CREDIT / COMPTEUR
; Y1  PILOTAGGIO POMPA
; Y2..Y7 PROGRAM 1..6
; Y23 RM5 INHIBIT
; Y27 LIGHTING
;
; HMI:
; M400 STOP
; M401..M406 P1..P6
; M410 PILOTAGGIO_ENABLE
; M411 RESET_RM5_SESSION_COUNTERS
; M413 TIME_PARAM_OK
;
; Internal:
; M300 AUTOMATE_PRESENT
; M301 PRACA
; M302 MANUAL_FREE status
; M320 RM5 one-scan pulse
; M321 RM5 packet active
; M330 CREDIT phase ON
; M331 CREDIT phase OFF
; M412 WORK_ACTIVE
; M420..M425 P1..P6 ACTIVE
; M440 STOP_CMD
; M441..M446 P1..P6 CMD
;
; Data:
; D300 RM5 CH1 multiplier/value, startup = 10
; D320 current RM5 packet pulse count
; D330 queued CREDIT pulses
; D350 current value / time
; D351 reserved/high word
; D520 seconds per RM5 pulse, HMI RW, startup = 10
; D521 last packet pulse count
; D522:D523 last packet converted time (s)
; D524 RM5 session pulse counter
; D525:D526 session equivalent time (s), 32-bit
; D527 saturation threshold = 32000 - D520
; D500..D507 PLC version string for HMI
; D516 VERSION_MAJOR
; D517 VERSION_MINOR
; D518 VERSION_PATCH
;
; Timers:
; T200 K100 packet end timeout (~1 s for 10 ms base)
; T201 K10 CREDIT ON (~0.1 s)
; T202 K10 CREDIT OFF (~0.1 s)

; ============================================================
; 3. RM5 INHIBIT
; Missing AUTOMATE_PRESENT blocks RM5.
; ============================================================

OUT Y23


; ============================================================
; 4. RM5 CH1 EDGE DETECTION AND PACKET COUNTING
; X27 is counted once per rising edge.
; If actual hardware polarity is inverted, replace LDP with LDF.
; ============================================================

LDP X27
OUT M320

LD M320
INC D320

LD M320
AND< D524 K30000
INC D524

LD M413
SUB K32000 D520 D527

LD M320
AND M413
AND<= D350 D527
ADD D350 D520 D350

LD M320
AND M413
AND> D350 D527
MOV K32000 D350

LD M320
SET M321

LD M320
RST T200

LD M321
OUT T200 K100


; ============================================================
; 5. END OF RM5 PACKET
; ============================================================
